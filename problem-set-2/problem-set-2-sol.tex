\section{Problems Set 2}
\begin{problem}[25pt]
  Let $a_n$ and $b_n$ be a sequence of real numbers.
  \begin{enumerate}
    \item Assume that $\lim_{n \to \infty} a_n$ and $\lim_{n \to \infty} b_n$ exist. Show that
      \[
        \lim_{n \to \infty} (a_n b_n)
        = \left( \lim_{n \to \infty} a_n \right) \left( \lim_{n \to \infty} b_n \right).
      \]
    \item Give an example in which $\lim_{n \to \infty} (a_n b_n)$ exists but neither
      $\lim_{n \to \infty} a_n$ nor $\lim_{n \to \infty} b_n$ exists.
  \end{enumerate}
\end{problem}

\begin{problem}[25pt]
  Find the limit for the following sequence if it exists. Or show that the limit doesn't
  exist.
  \begin{enumerate}
    \item $a_n = \dfrac{n^2}{n+1} - \dfrac{n^2+1}{n}$
    \item $a_n = \dfrac{\sin(n)}{n}$
    \item \[
        a_n = \underbrace{\frac{n^2}{\sqrt{n^6+1}} + \frac{n^2}{\sqrt{n^6+2}}
          + \frac{n^2}{\sqrt{n^6+3}} + \dots + \frac{n^2}{\sqrt{n^6+n}}}_{n \text{ terms}}
      \]
  \end{enumerate}
\end{problem}

\begin{problem}[15pt]
  Let $a_n$ be a sequence of real numbers and $L$ be a real number. Show that the following
  two statements are equivalent. One holds if and only if the other does.
  \begin{itemize}
    \item There exists a subsequence $a_{n_k}$ converging to $L$.
    \item For any $\epsilon > 0$, there exist infinite any $a_n$ in $(L - \epsilon, L + \epsilon)$.
  \end{itemize}
\end{problem}

\begin{proof} (Forward direction)
  $\Longrightarrow$ Assume that there exists subsequence $a_{n_k}$ such that $a_{n_k} \rightarrow L$. 
  Then for any $\epsilon > 0$, there exists positive integer $K$ such that for all $k \geq K$ $$|a_{n_k} - L| < \epsilon$$
  then $L-\epsilon < a_{n_k} < L+\epsilon$ for all $k \geq K$. There are infinitely many $a_n$ such that $a_n \in (L-\epsilon,L+\epsilon)$ for all $\epsilon > 0$ 
\end{proof}

\begin{proof} (Backward direction)
  $\Longleftarrow$ Assume that for all $\epsilon > 0$ there exists infinite $a_n$ such that $a_n \in (L-\epsilon,L+\epsilon)$
 then for each $k \in \NN$ there are infinitely many $i$ such that $a_i \in (L-\frac1k,L+\frac1k)$ then for each $k$, we choose $n_k$ such that $n_k > n_{k-1}$ and $|a_{n_k}-L| \leq \frac1k$
 for each $\epsilon > 0$ there exists some integer $K$ such that $K > \frac1\epsilon$ (Archimedean Property) so that for all $k \geq K$ 
 \[
 |a_{n_k} - L| \leq \frac1k \leq \frac1K  < \epsilon
 \]
 Hence, the subsequence $a_{n_k}$ converges to $L$
\end{proof}
